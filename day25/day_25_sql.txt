DAY 25 — DATABASE FUNDAMENTALS & CRUD
SQL USED IN THE N8N WORKFLOW

============================================================
1. CREATE TABLE
============================================================

CREATE TABLE IF NOT EXISTS public.tickets (
    ticket_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    customer_name TEXT NOT NULL,
    email TEXT NOT NULL,
    category TEXT NOT NULL,
    priority TEXT NOT NULL,
    status TEXT NOT NULL,
    ai_summary TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


============================================================
2. CREATE — INSERT DEMO TICKETS
============================================================

INSERT INTO public.tickets
    (customer_name, email, category, priority, status, ai_summary)
VALUES
    ('Ali Khan', 'ali@example.com', 'Login Issue', 'High', 'Open',
     'Customer cannot log into the application.'),
    ('Sara Ahmed', 'sara@example.com', 'Billing Issue', 'Medium', 'Open',
     'Customer reports an incorrect charge and asked for clarification.')
RETURNING ticket_id, customer_name, email, category, priority, status, ai_summary, created_at;


============================================================
3. READ — GET ALL TICKETS
============================================================

SELECT
    ticket_id,
    customer_name,
    email,
    category,
    priority,
    status,
    ai_summary,
    created_at
FROM public.tickets
ORDER BY ticket_id;


============================================================
4. UPDATE — CHANGE TICKET STATUS
============================================================

UPDATE public.tickets
SET status = 'Resolved'
WHERE ticket_id = $1
RETURNING ticket_id, customer_name, status;

NOTE:
In n8n, $1 is supplied by the ticket_id produced by the CREATE node.


============================================================
5. READ — CHECK THE UPDATED TICKET
============================================================

SELECT
    ticket_id,
    customer_name,
    email,
    category,
    priority,
    status,
    ai_summary,
    created_at
FROM public.tickets
WHERE ticket_id = $1;

NOTE:
In n8n, $1 is supplied by the ticket_id produced by the CREATE node.


============================================================
6. DELETE — DELETE THE TEST TICKET
============================================================

DELETE FROM public.tickets
WHERE ticket_id = $1
RETURNING ticket_id, customer_name, email, status;

NOTE:
In the workflow, this deletes the second demo ticket.


============================================================
7. FINAL READ — VERIFY THE DATABASE
============================================================

SELECT
    ticket_id,
    customer_name,
    email,
    category,
    priority,
    status,
    ai_summary,
    created_at
FROM public.tickets
ORDER BY ticket_id;


============================================================
CRUD MAPPING
============================================================

CREATE  -> INSERT
READ    -> SELECT
UPDATE  -> UPDATE
DELETE  -> DELETE

CREATE TABLE is DDL, not CRUD.
